<objective>
Add booking-participant validation to the /api/notifications route so only confirmed booking participants can send notifications referencing a booking.

Purpose: Currently any authenticated user can POST to /api/notifications with any bookingId and any userId as the recipient. This allows notification spoofing — a malicious user can impersonate a booking participant and spam or deceive another user. The fix validates that the caller is either the client_id or provider_id of the booking before the insert proceeds.

Output:
- src/app/api/notifications/route.ts: booking-participant guard inserted after the missing-fields check and before the admin insert
</objective>

import { createSupabaseServerClient, createSupabaseAdminClient } from '@/lib/supabase-server'

export async function POST(request: NextRequest) {
  // Require a valid logged-in user to prevent abuse
  const supabase = await createSupabaseServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { userId, type, message, bookingId } = await request.json()
  if (!userId || !type || !message) {
    return NextResponse.json({ error: 'Missing fields' }, { status: 400 })
  }

  // Admin client bypasses RLS — reused for both participant check and insert
  const admin = createSupabaseAdminClient()

  // Booking-participant guard: if bookingId is provided, caller must be client or provider
  if (bookingId) {
    const { data: booking } = await admin
      .from('bookings')
      .select('client_id, provider_id')
      .eq('id', bookingId)
      .single()

    if (!booking || (booking.client_id !== user.id && booking.provider_id !== user.id)) {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
    }
  }

  const { error } = await admin.from('notifications').insert({
    user_id: userId,
    type,
    message,
    ...(bookingId ? { booking_id: bookingId } : {}),
  })
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })

  return NextResponse.json({ ok: true })
}
